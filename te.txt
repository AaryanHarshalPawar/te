int* sortArrayByParity(int* nums, int numsSize, int* returnSize) {
    int i = 0, j = numsSize - 1, temp;

    while (i < j) {
        if (nums[i] % 2 == 0)
            i++;
        else if (nums[j] % 2 == 1)
            j--;
        else {
            temp = nums[i];
            nums[i] = nums[j];
            nums[j] = temp;
            i++;
            j--;
        }
    }

    *returnSize = numsSize;
    return nums;
}

#include <stdio.h>

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int partition(int arr[], int low, int high) {
    int pivot = arr[high];
    int i = low - 1;
    int j;

    for (j = low; j < high; j++) {
        if (arr[j] < pivot) {
            i++;
            swap(&arr[i], &arr[j]);
        }
    }

    swap(&arr[i + 1], &arr[high]);
    return i + 1;
}

void quickSort(int arr[], int low, int high) {
    int p;

    if (low < high) {
        p = partition(arr, low, high);
        quickSort(arr, low, p - 1);
        quickSort(arr, p + 1, high);
    }
}

#include <stdio.h>

void merge(int arr[], int left, int mid, int right) {
    int temp[100];
    int i = left, j = mid + 1, k = left;

    while (i <= mid && j <= right) {
        if (arr[i] <= arr[j])
            temp[k++] = arr[i++];
        else
            temp[k++] = arr[j++];
    }

    while (i <= mid)
        temp[k++] = arr[i++];

    while (j <= right)
        temp[k++] = arr[j++];

    for (i = left; i <= right; i++)
        arr[i] = temp[i];
}

void mergeSort(int arr[], int left, int right) {
    int mid;

    if (left < right) {
        mid = (left + right) / 2;
        mergeSort(arr, left, mid);
        mergeSort(arr, mid + 1, right);
        merge(arr, left, mid, right);
    }
}

/* ---- A3 | Code 5: Problem 1: 912. Sort an Array (Quick Sort) ---- */
#include <stdlib.h>

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

void quickSort(int* nums, int low, int high) {
    int pivot, lt, gt, i;

    if (low >= high)
        return;

    pivot = nums[low + rand() % (high - low + 1)];
    lt = low;
    gt = high;
    i = low;

    while (i <= gt) {
        if (nums[i] < pivot) {
            swap(&nums[i], &nums[lt]);
            lt++;
            i++;
        }
        else if (nums[i] > pivot) {
            swap(&nums[i], &nums[gt]);
            gt--;
        }
        else
            i++;
    }

    quickSort(nums, low, lt - 1);
    quickSort(nums, gt + 1, high);
}

int* sortArray(int* nums, int numsSize, int* returnSize) {
    quickSort(nums, 0, numsSize - 1);
    *returnSize = numsSize;
    return nums;
}

#include <stdlib.h>

void merge(int* nums, int left, int mid, int right, int* temp) {
    int i = left, j = mid + 1, k = left;

    while (i <= mid && j <= right) {
        if (nums[i] <= nums[j])
            temp[k++] = nums[i++];
        else
            temp[k++] = nums[j++];
    }

    while (i <= mid)
        temp[k++] = nums[i++];

    while (j <= right)
        temp[k++] = nums[j++];

    for (i = left; i <= right; i++)
        nums[i] = temp[i];
}

void mergeSort(int* nums, int left, int right, int* temp) {
    int mid;

    if (left < right) {
        mid = left + (right - left) / 2;
        mergeSort(nums, left, mid, temp);
        mergeSort(nums, mid + 1, right, temp);
        merge(nums, left, mid, right, temp);
    }
}

int* sortArray(int* nums, int numsSize, int* returnSize) {
    int* temp = (int*)malloc(numsSize * sizeof(int));

    mergeSort(nums, 0, numsSize - 1, temp);
    free(temp);

    *returnSize = numsSize;
    return nums;
}

/* ---- A4 | Code 1: Part A: Insertion Operations ---- */
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *head = NULL;

void insertAtBeginning(int value) {
    struct Node *newNode = (struct Node *)malloc(sizeof(struct Node));

    newNode->data = value;
    newNode->next = head;
    head = newNode;

    printf("%d inserted at the beginning\n", value);
}

void insertAtEnd(int value) {
    struct Node *newNode = (struct Node *)malloc(sizeof(struct Node));
    struct Node *temp = head;

    newNode->data = value;
    newNode->next = NULL;

    if (head == NULL) {
        head = newNode;
    }
    else {
        while (temp->next != NULL)
            temp = temp->next;

        temp->next = newNode;
    }

    printf("%d inserted at the end\n", value);
}

void insertAtPosition(int value, int pos) {
    struct Node *newNode, *temp = head;
    int i;

    if (pos < 1) {
        printf("Invalid position\n");
        return;
    }

    if (pos == 1) {
        insertAtBeginning(value);
        return;
    }

    for (i = 1; i < pos - 1 && temp != NULL; i++)
        temp = temp->next;

    if (temp == NULL) {
        printf("Position out of range\n");
        return;
    }

    newNode = (struct Node *)malloc(sizeof(struct Node));
    newNode->data = value;
    newNode->next = temp->next;
    temp->next = newNode;

    printf("%d inserted at position %d\n", value, pos);
}

void display() {
    struct Node *temp = head;

    if (head == NULL) {
        printf("List is empty\n");
        return;
    }

    printf("List: ");
    while (temp != NULL) {
        printf("%d -> ", temp->data);
        temp = temp->next;
    }
    printf("NULL\n");
}

int main() {
    int choice, value, pos;

    while (1) {
        printf("\n1. Insert at Beginning\n");
        printf("2. Insert at End\n");
        printf("3. Insert at Position\n");
        printf("4. Display\n");
        printf("5. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        if (choice == 1) {
            printf("Enter value: ");
            scanf("%d", &value);
            insertAtBeginning(value);
        }
        else if (choice == 2) {
            printf("Enter value: ");
            scanf("%d", &value);

/* ---- A4 | Code 5: Problem 1. 203. Remove Linked List Elements ---- */
#include <stdio.h>
#include <stdlib.h>

struct ListNode {
    int val;
    struct ListNode *next;
};

struct ListNode* removeElements(struct ListNode* head, int val) {
    struct ListNode *current, *temp;

    while (head != NULL && head->val == val) {
        temp = head;
        head = head->next;
        free(temp);
    }

    current = head;

    while (current != NULL && current->next != NULL) {
        if (current->next->val == val) {
            temp = current->next;
            current->next = temp->next;
            free(temp);
        }
        else {
            current = current->next;
        }
    }

    return head;
}

struct ListNode* buildList(int arr[], int n) {
    struct ListNode *head = NULL, *tail = NULL, *newNode;
    int i;

    for (i = 0; i < n; i++) {
        newNode = (struct ListNode *)malloc(sizeof(struct ListNode));
        newNode->val = arr[i];
        newNode->next = NULL;

        if (head == NULL)
            head = newNode;
        else
            tail->next = newNode;

        tail = newNode;
    }

    return head;
}

void printList(struct ListNode *head) {
    while (head != NULL) {
        printf("%d -> ", head->val);
        head = head->next;
    }
    printf("NULL\n");
}

void freeList(struct ListNode *head) {
    struct ListNode *temp;

    while (head != NULL) {
        temp = head;
        head = head->next;
        free(temp);
    }
}

int main() {
    int arr[] = {1, 2, 6, 3, 4, 5, 6};
    int n = 7, val = 6;
    struct ListNode *head = buildList(arr, n);

    printf("Original list: ");
    printList(head);

    head = removeElements(head, val);

    printf("After removing %d: ", val);
    printList(head);

    freeList(head);

    return 0;
}

#include <stdio.h>
#include <stdlib.h>

struct ListNode {
    int val;
    struct ListNode *next;
};

struct ListNode* reverseList(struct ListNode* head) {
    struct ListNode *prev = NULL, *current = head, *next = NULL;

    while (current != NULL) {
        next = current->next;
        current->next = prev;
        prev = current;
        current = next;
    }

    return prev;
}

struct ListNode* buildList(int arr[], int n) {

    for (i = 0; i < n; i++) {

        newNode->next = NULL;

            head = newNode;

/* ---- A5 | Code 5: Problem 1. 430. Flatten a Multilevel Doubly Linked List ---- */
class Solution {
public:
    Node* flatten(Node* head) {
        Node* current = head;

        while (current != nullptr) {
            if (current->child != nullptr) {
                Node* next = current->next;
                Node* tail = current->child;

                while (tail->next != nullptr)
                    tail = tail->next;

                current->next = current->child;
                current->child->prev = current;
                current->child = nullptr;

                tail->next = next;
                if (next != nullptr)
                    next->prev = tail;
            }

            current = current->next;
        }

        return head;
    }
};

        printf("No processes in the queue\n");

        if (head->id != id) {

        free(head);
        head = NULL;

        head = head->next;
        prev->next = head;

#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

struct ListNode {
    int val;
    struct ListNode *next;
};

bool hasCycle(struct ListNode *head) {
    struct ListNode *slow = head, *fast = head;

    while (fast != NULL && fast->next != NULL) {
        slow = slow->next;
        fast = fast->next->next;

        if (slow == fast)
            return true;
    }

    return false;
}

struct ListNode* buildList(int arr[], int n, int pos) {
    struct ListNode *head = NULL, *tail = NULL, *cycleNode = NULL, *newNode;
    int i;

    for (i = 0; i < n; i++) {
        newNode = (struct ListNode *)malloc(sizeof(struct ListNode));
        newNode->val = arr[i];
        newNode->next = NULL;

        if (head == NULL)
            head = newNode;
        else
            tail->next = newNode;

        tail = newNode;

        if (i == pos)
            cycleNode = newNode;
    }

    if (pos >= 0 && tail != NULL)
        tail->next = cycleNode;

    return head;
}

void freeList(struct ListNode *head, int n) {
    struct ListNode *temp;
    int i;

    for (i = 0; i < n && head != NULL; i++) {
        temp = head;
        head = head->next;
        free(temp);
    }
}

int main() {
    int arr1[] = {3, 2, 0, -4};
    int arr2[] = {1, 2};
    int arr3[] = {1};

    struct ListNode *list1 = buildList(arr1, 4, 1);
    struct ListNode *list2 = buildList(arr2, 2, 0);
    struct ListNode *list3 = buildList(arr3, 1, -1);

    printf("List [3,2,0,-4] with tail joined to index 1\n");

    printf("Has cycle: %s\n\n", hasCycle(list2) ? "true" : "false");

    printf("List [1] with no cycle\n");
    printf("Has cycle: %s\n", hasCycle(list3) ? "true" : "false");

    freeList(list2, 2);
    freeList(list3, 1);

    return 0;
}

    while (fast != NULL && fast->next != NULL) {
        slow = slow->next;
        fast = fast->next->next;

        if (slow == fast) {
            slow = head;

            while (slow != fast) {
                slow = slow->next;
                fast = fast->next;
            }

            return slow;
        }
    }

    return NULL;
}

    obj->data[obj->top] = val;

    if (obj->top == 0 || val < obj->mins[obj->top - 1])

    else
        obj->mins[obj->top] = obj->mins[obj->top - 1];

/* ---- A7 | Code 6: Problem 2. 622. Design Circular Queue ---- */
typedef struct {
    int *data;
    int front;
    int count;
    int size;
} MyCircularQueue;

MyCircularQueue* myCircularQueueCreate(int k) {
    MyCircularQueue *obj = (MyCircularQueue *)malloc(sizeof(MyCircularQueue));

    obj->data = (int *)malloc(k * sizeof(int));
    obj->front = 0;
    obj->count = 0;
    obj->size = k;

    return obj;
}

bool myCircularQueueEnQueue(MyCircularQueue* obj, int value) {
    if (obj->count == obj->size)
        return false;

    obj->data[(obj->front + obj->count) % obj->size] = value;
    obj->count++;

    return true;
}

bool myCircularQueueDeQueue(MyCircularQueue* obj) {
    if (obj->count == 0)
        return false;

    obj->front = (obj->front + 1) % obj->size;
    obj->count--;

    return true;
}

int myCircularQueueFront(MyCircularQueue* obj) {
    if (obj->count == 0)
        return -1;

    return obj->data[obj->front];
}

int myCircularQueueRear(MyCircularQueue* obj) {
    if (obj->count == 0)
        return -1;

    return obj->data[(obj->front + obj->count - 1) % obj->size];
}

bool myCircularQueueIsEmpty(MyCircularQueue* obj) {
    return obj->count == 0;
}

bool myCircularQueueIsFull(MyCircularQueue* obj) {
    return obj->count == obj->size;
}

void myCircularQueueFree(MyCircularQueue* obj) {
    free(obj->data);
    free(obj);
}

    return c == '+' || c == '-' || c == '*' || c == '/' || c == '%' || c == '^';
}

int precedence(char op) {
    if (op == '^')
        return 3;
    else if (op == '*' || op == '/' || op == '%')
        return 2;
    else if (op == '+' || op == '-')
        return 1;
    else
        return 0;
}

int main() {
    char infix[100], postfix[100];
    char c;
    int i, j = 0;

    printf("Enter infix expression: ");
    fgets(infix, sizeof(infix), stdin);

    for (i = 0; infix[i] != '\0' && infix[i] != '\n'; i++) {
        c = infix[i];

        if (c == ' ')
            continue;

        if (isalnum(c)) {
            postfix[j] = c;
            j++;
        }
        else if (c == '(') {
            push(c);
        }
        else if (c == ')') {
            while (!isEmpty() && peek() != '(') {
                postfix[j] = pop();
                j++;
            }

            if (isEmpty()) {
                printf("Invalid expression, parentheses are not balanced\n");
                return 0;
            }

            pop();
        }
        else if (isOperator(c)) {
            while (!isEmpty() && peek() != '(') {
                if (precedence(peek()) > precedence(c) ||
                    (precedence(peek()) == precedence(c) && c != '^')) {
                    postfix[j] = pop();
                    j++;
                }
                else {
                    break;
                }
            }

            push(c);
        }
        else {
            printf("Invalid character %c in the expression\n", c);
            return 0;
        }
    }

    while (!isEmpty()) {
        if (peek() == '(') {
            printf("Invalid expression, parentheses are not balanced\n");
            return 0;
        }

        postfix[j] = pop();
        j++;
    }

    postfix[j] = '\0';

    printf("Postfix expression: %s\n", postfix);

    return 0;
}

#include <stdio.h>

#define INF 9999

int main() {
    int n, e, i, j, u, v, w, k;
    int cost[20][20], selected[20], minw[20], parent[20];
    int total = 0, min;

    printf("Enter the number of vertices: ");
    scanf("%d", &n);
    printf("Enter the number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < n; i++) {
        selected[i] = 0;
        minw[i] = INF;
        parent[i] = -1;
        for (j = 0; j < n; j++)
            cost[i][j] = 0;
    }

    for (i = 0; i < e; i++) {
        printf("Enter edge %d (u v weight): ", i + 1);
        scanf("%d %d %d", &u, &v, &w);
        cost[u][v] = w;
        cost[v][u] = w;
    }

    minw[0] = 0;

    for (k = 0; k < n; k++) {
        min = INF;
        u = -1;
        for (i = 0; i < n; i++) {
            if (selected[i] == 0 && minw[i] < min) {
                min = minw[i];
                u = i;
            }
        }

        if (u == -1) {
            printf("\nGraph is not connected.\n");
            return 0;
        }

        selected[u] = 1;

        for (v = 0; v < n; v++) {
            if (cost[u][v] != 0 && selected[v] == 0 && cost[u][v] < minw[v]) {
                minw[v] = cost[u][v];
                parent[v] = u;
            }
        }
    }

    printf("\nEdges in Minimum Spanning Tree:\n");
    printf("Edge\tWeight\n");
    for (i = 1; i < n; i++) {
        printf("%d - %d\t%d\n", parent[i], i, cost[i][parent[i]]);
        total = total + cost[i][parent[i]];
    }
    printf("Total weight of MST = %d\n", total);

    return 0;
}

/* ---- A18 | Code 1: C Program for Floyd–Warshall Algorithm ---- */
#include <stdio.h>

#define INF 99999

int main() {
    int n, e, i, j, k, u, v, w;
    int dist[20][20];
    int negative = 0;

    printf("Enter the number of vertices: ");
    scanf("%d", &n);
    printf("Enter the number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            if (i == j)
                dist[i][j] = 0;
            else
                dist[i][j] = INF;
        }
    }

    for (i = 0; i < e; i++) {
        printf("Enter edge %d (u v weight): ", i + 1);
        scanf("%d %d %d", &u, &v, &w);
        dist[u][v] = w;
    }

    for (k = 0; k < n; k++) {
        for (i = 0; i < n; i++) {
            for (j = 0; j < n; j++) {
                if (dist[i][k] != INF && dist[k][j] != INF) {
                    if (dist[i][k] + dist[k][j] < dist[i][j])
                        dist[i][j] = dist[i][k] + dist[k][j];
                }
            }
        }
    }

    for (i = 0; i < n; i++) {
        if (dist[i][i] < 0)
            negative = 1;
    }

    if (negative == 1) {
        printf("\nThe graph contains a negative weight cycle.\n");
        return 0;
    }

    printf("\nShortest Distance Matrix:\n");
    printf("     ");
    for (i = 0; i < n; i++)
        printf("%5d", i);
    printf("\n");
    for (i = 0; i < n; i++) {
        printf("%5d", i);
        for (j = 0; j < n; j++) {
            if (dist[i][j] == INF)
                printf("%5s", "INF");
            else
                printf("%5d", dist[i][j]);
        }
        printf("\n");
    }

    return 0;
}

#include <stdio.h>
#include <stdlib.h>

struct node {
    int key;
    struct node *next;
};

struct node *table[20];
int size;

void insert(int key) {
    int index = key % size;
    struct node *newnode, *temp;
    newnode = (struct node *)malloc(sizeof(struct node));
    newnode->key = key;
    newnode->next = NULL;
    if (table[index] == NULL) {
        table[index] = newnode;
    } else {
        temp = table[index];
        while (temp->next != NULL)
            temp = temp->next;
        temp->next = newnode;
    }
    printf("Key %d inserted at bucket %d\n", key, index);
}

void search(int key) {
    int index = key % size;
    int pos = 1;
    struct node *temp = table[index];
    while (temp != NULL) {
        if (temp->key == key) {
            printf("Key %d found in bucket %d at position %d\n", key, index, pos);
            return;
        }
        temp = temp->next;
        pos++;
    }
    printf("Key %d not found\n", key);
}

void deleteKey(int key) {
    int index = key % size;
    struct node *temp = table[index];
    struct node *prev = NULL;
    while (temp != NULL) {
        if (temp->key == key) {
            if (prev == NULL)
                table[index] = temp->next;
            else
                prev->next = temp->next;
            free(temp);
            printf("Key %d deleted from bucket %d\n", key, index);
            return;
        }
        prev = temp;
        temp = temp->next;
    }
    printf("Key %d not found\n", key);
}

void display() {
    int i;
    struct node *temp;
    printf("\nHash Table:\n");
    for (i = 0; i < size; i++) {

        temp = table[i];

            temp = temp->next;
        }

    }
}

/* ---- A20 | Code 2: Performance Comparison Experiment: Separate Chaining versus Linear Probing ---- */
#include <stdio.h>
#include <stdlib.h>

#define SIZE 11

struct node {
    int key;
    struct node *next;
};

struct node *chain[SIZE];
int probe[SIZE];
int chainCollisions = 0, probeCollisions = 0;

void insertChain(int key) {
    int index = key % SIZE;
    struct node *newnode, *temp;
    newnode = (struct node *)malloc(sizeof(struct node));
    newnode->key = key;
    newnode->next = NULL;
    if (chain[index] == NULL) {
        chain[index] = newnode;
    } else {
        chainCollisions++;
        temp = chain[index];
        while (temp->next != NULL)
            temp = temp->next;
        temp->next = newnode;
    }
}

void insertProbe(int key) {
    int index = key % SIZE;
    if (probe[index] != -1)
        probeCollisions++;
    while (probe[index] != -1)
        index = (index + 1) % SIZE;
    probe[index] = key;
}

int searchChain(int key) {
    int count = 0;
    struct node *temp = chain[key % SIZE];
    while (temp != NULL) {
        count++;
        if (temp->key == key)
            return count;
        temp = temp->next;
    }
    return count;
}

int searchProbe(int key) {
    int index = key % SIZE;
    int count = 0, i = 0;
    while (probe[index] != -1 && i < SIZE) {
        count++;
        if (probe[index] == key)
            return count;
        index = (index + 1) % SIZE;
        i++;
    }
    return count + 1;
}

int main() {
    int keys[8] = {21, 32, 43, 54, 65, 76, 87, 18};
    int searchKeys[4] = {21, 87, 18, 99};
    int i;
    struct node *temp;

    for (i = 0; i < SIZE; i++) {
        chain[i] = NULL;
        probe[i] = -1;
    }

    for (i = 0; i < 8; i++) {
        insertChain(keys[i]);
        insertProbe(keys[i]);
    }

    printf("Separate Chaining Table:\n");
    for (i = 0; i < SIZE; i++) {
        printf("%2d: ", i);
        temp = chain[i];
        while (temp != NULL) {
            printf("%d -> ", temp->key);
            temp = temp->next;
        }
        printf("NULL\n");
    }

    printf("\nLinear Probing Table:\n");
    for (i = 0; i < SIZE; i++) {
        if (probe[i] == -1)
            printf("%2d: -\n", i);
        else
            printf("%2d: %d\n", i, probe[i]);
    }

    printf("\nCollisions during insertion:\n");
    printf("Separate Chaining = %d\n", chainCollisions);
    printf("Linear Probing    = %d\n", probeCollisions);

    printf("\nKey comparisons for search:\n");
    printf("Key\tChaining\tLinear Probing\n");
    for (i = 0; i < 4; i++)
        printf("%d\t%d\t\t%d\n", searchKeys[i], searchChain(searchKeys[i]), searchProbe(searchKeys[i]));

    return 0;
}
